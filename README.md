const passing = new Date('2026-07-31T00:00:00');
function update(){
  const now = Date.now();
  const elapsed = Math.max(0, now - passing.getTime());
  const total = Math.floor(elapsed/1000);
  const d = Math.floor(total/86400);
  const h = Math.floor((total%86400)/3600);
  const m = Math.floor((total%3600)/60);
  const s = total%60;
  document.getElementById('days').textContent=d.toLocaleString();
  document.getElementById('hours').textContent=String(h).padStart(2,'0');
  document.getElementById('minutes').textContent=String(m).padStart(2,'0');
  document.getElementById('seconds').textContent=String(s).padStart(2,'0');

  const today = new Date();
  let target = new Date(today.getFullYear(),6,31,0,0,0);
  if(target.getTime() <= now) target = new Date(today.getFullYear()+1,6,31,0,0,0);
  const left = Math.max(0, Math.floor((target.getTime()-now)/1000));
  const nd=Math.floor(left/86400), nh=Math.floor((left%86400)/3600), nm=Math.floor((left%3600)/60), ns=left%60;
  document.getElementById('nextCount').textContent =
    `${nd.toLocaleString()} Days · ${String(nh).padStart(2,'0')} Hours · ${String(nm).padStart(2,'0')} Minutes · ${String(ns).padStart(2,'0')} Seconds`;
}
update();
setInterval(update,1000);
</script>
